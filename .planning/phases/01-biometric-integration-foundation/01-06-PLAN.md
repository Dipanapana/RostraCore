**Create `frontend/src/lib/biometric/geolocation.ts`:**

```typescript
export interface GPSPosition {
  lat: number;
  lng: number;
  accuracy: number; // meters
}

export async function getCurrentPosition(): Promise<GPSPosition> {
  /**
   * Get current GPS position with high accuracy.
   * Rejects if:
   * - Geolocation not supported
   * - User denies permission
   * - Timeout (10 seconds)
   * - Accuracy > 50m (too imprecise per research)
   */
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) {
      reject(new Error('Geolocation not supported'));
      return;
    }
    navigator.geolocation.getCurrentPosition(
      (position) => {
        const { latitude, longitude, accuracy } = position.coords;
        if (accuracy > 50) {
          reject(new Error(`GPS accuracy too low (${Math.round(accuracy)}m). Move to open area or wait for better signal.`));
          return;
        }
        resolve({ lat: latitude, lng: longitude, accuracy });
      },
      (error) => {
        const messages: Record<number, string> = {
          1: 'Location permission denied. Please enable in browser settings.',
          2: 'Location unavailable. Check GPS/WiFi settings.',
          3: 'Location request timed out. Try again in open area.',
        };
        reject(new Error(messages[error.code] || 'Unknown geolocation error'));
      },
      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
    );
  });
}

export function checkGPSAccuracy(accuracy: number): 'good' | 'acceptable' | 'poor' {
  if (accuracy <= 15) return 'good';
  if (accuracy <= 50) return 'acceptable';
  return 'poor';
}
```

  - Only < 70% is actual failure
