    ```python
    """Exchange rate service with API fetching, DB caching, and fallback."""

    import httpx
    from datetime import datetime, timedelta, timezone
    from decimal import Decimal
    from sqlalchemy.orm import Session
    from app.models.country_config import ExchangeRate
    import logging

    logger = logging.getLogger(__name__)

    class ExchangeRateService:
        """Fetch, cache, and provide exchange rates."""

        CACHE_DURATION_HOURS = 4  # Refresh cached rates every 4 hours

        # Free APIs, no API key needed
        PRIMARY_API = "https://open.er-api.com/v6/latest/{base}"
        FALLBACK_API = "https://api.frankfurter.dev/v1/latest?base={base}"

        @classmethod
        async def get_rate(
            cls,
            from_currency: str,
            to_currency: str,
            db: Session
        ) -> Decimal:
            """Get exchange rate with caching and fallback.

            Strategy:
            1. Check DB cache (if fresh enough, return cached)
            2. Try primary API (ExchangeRate-API)
            3. Try fallback API (Frankfurter)
            4. Return stale cached rate if all APIs fail
            5. Raise if no rate available at all

            Args:
                from_currency: Source currency code (e.g., "ZAR")
                to_currency: Target currency code (e.g., "USD")
                db: Database session

            Returns:
                Exchange rate as Decimal
            """
            if from_currency == to_currency:
                return Decimal("1")

            # Check cache
            cached = cls._get_cached_rate(from_currency, to_currency, db)
            if cached and not cls._is_stale(cached):
                return Decimal(str(cached.rate))

            # Try primary API
            rate = None
            try:
                rate = await cls._fetch_from_primary(from_currency, to_currency)
            except Exception as e:
                logger.warning(f"Primary API failed for {from_currency}->{to_currency}: {e}")

            # Try fallback API
            if rate is None:
                try:
                    rate = await cls._fetch_from_fallback(from_currency, to_currency)
                except Exception as e:
                    logger.warning(f"Fallback API failed for {from_currency}->{to_currency}: {e}")

            # Use API rate if available
            if rate is not None:
                cls._cache_rate(from_currency, to_currency, rate, "api", db)
                return rate

            # Return stale cache if available
            if cached:
                logger.warning(
                    f"Using stale rate for {from_currency}->{to_currency} "
                    f"(fetched {cached.fetched_at})"
                )
                return Decimal(str(cached.rate))

            raise ExchangeRateUnavailable(
                f"No exchange rate available for {from_currency} -> {to_currency}"
            )

        @classmethod
        async def convert(
            cls,
            amount: Decimal,
            from_currency: str,
            to_currency: str,
            db: Session
        ) -> tuple[Decimal, Decimal]:
            """Convert amount between currencies.

            Returns:
                Tuple of (converted_amount, rate_used)
            """
            rate = await cls.get_rate(from_currency, to_currency, db)
            converted = (amount * rate).quantize(Decimal("0.01"))
            return converted, rate

        @classmethod
        async def lock_rate_for_payroll(
            cls,
            from_currency: str,
            to_currency: str,
            db: Session
        ) -> Decimal:
            """Lock exchange rate for payroll generation.

            Same as get_rate but logged as "locked for payroll" for audit trail.
            The caller should store this rate with the payroll record.
            """
            rate = await cls.get_rate(from_currency, to_currency, db)
            logger.info(
                f"Locked exchange rate for payroll: "
                f"{from_currency}->{to_currency} = {rate}"
            )
            return rate

        @classmethod
        def set_manual_rate(
            cls,
            from_currency: str,
            to_currency: str,
            rate: Decimal,
            db: Session
        ) -> ExchangeRate:
            """Set a manual exchange rate (for admin override)."""
            return cls._cache_rate(from_currency, to_currency, rate, "manual", db)

        @classmethod
        def _get_cached_rate(cls, from_cur, to_cur, db):
            return db.query(ExchangeRate).filter(
                ExchangeRate.base_currency == from_cur,
                ExchangeRate.target_currency == to_cur
            ).order_by(ExchangeRate.fetched_at.desc()).first()

        @classmethod
        def _is_stale(cls, cached_rate):
            if not cached_rate or not cached_rate.fetched_at:
                return True
            stale_threshold = datetime.now(timezone.utc) - timedelta(hours=cls.CACHE_DURATION_HOURS)
            fetched = cached_rate.fetched_at
            if fetched.tzinfo is None:
                fetched = fetched.replace(tzinfo=timezone.utc)
            return fetched < stale_threshold

        @classmethod
        async def _fetch_from_primary(cls, from_cur, to_cur):
            """Fetch from ExchangeRate-API (open endpoint, no key)."""
            url = cls.PRIMARY_API.format(base=from_cur)
            async with httpx.AsyncClient(timeout=10) as client:
                resp = await client.get(url)
                resp.raise_for_status()
                data = resp.json()
                if data.get("result") == "success":
                    rates = data.get("rates", {})
                    if to_cur in rates:
                        return Decimal(str(rates[to_cur]))
            raise ValueError(f"Rate not found in primary API response")

        @classmethod
        async def _fetch_from_fallback(cls, from_cur, to_cur):
            """Fetch from Frankfurter API (ECB-sourced, no key)."""
            url = cls.FALLBACK_API.format(base=from_cur)
            async with httpx.AsyncClient(timeout=10) as client:
                resp = await client.get(url)
                resp.raise_for_status()
                data = resp.json()
                rates = data.get("rates", {})
                if to_cur in rates:
                    return Decimal(str(rates[to_cur]))
            raise ValueError(f"Rate not found in fallback API response")

        @classmethod
        def _cache_rate(cls, from_cur, to_cur, rate, source, db):
            entry = ExchangeRate(
                base_currency=from_cur,
                target_currency=to_cur,
                rate=rate,
                source=source,
                fetched_at=datetime.now(timezone.utc)
            )
            db.add(entry)
            db.commit()
            return entry


    class ExchangeRateUnavailable(Exception):
        """Raised when no exchange rate is available."""
        pass
    ```

    ```python
    """Currency formatting and conversion utilities for backend."""

    from decimal import Decimal, ROUND_HALF_UP
    from typing import Optional

    # Try to import Babel for locale-aware formatting
    try:
        from babel.numbers import format_currency as babel_format_currency
        HAS_BABEL = True
    except ImportError:
        HAS_BABEL = False

    # Origin-country locale map (matches frontend CURRENCY_LOCALE_MAP)
    CURRENCY_LOCALE_MAP = {
        "ZAR": "en_ZA",
        "USD": "en_US",
        "GBP": "en_GB",
        "EUR": "de_DE",
        "NGN": "en_NG",
        "KES": "en_KE",
        "INR": "en_IN",
        "JPY": "ja_JP",
        "KWD": "ar_KW",
    }

    # Currency decimal places
    CURRENCY_DECIMALS = {
        "JPY": 0,
        "KWD": 3,
    }  # Default is 2

    class CurrencyService:
        """Backend currency formatting and arithmetic."""

        @staticmethod
        def format_money(amount: Decimal, currency_code: str) -> str:
            """Format a monetary amount using the currency's origin locale.

            Uses Babel if available, falls back to basic formatting.
            """
            locale = CURRENCY_LOCALE_MAP.get(currency_code, "en_US")

            if HAS_BABEL:
                return babel_format_currency(
                    float(amount), currency_code, locale=locale
                )

            # Fallback without Babel
            decimals = CURRENCY_DECIMALS.get(currency_code, 2)
            return f"{currency_code} {amount:.{decimals}f}"

        @staticmethod
        def round_to_currency(
            amount: Decimal,
            currency_code: str = "ZAR"
        ) -> Decimal:
            """Round amount to currency's minor unit precision."""
            decimals = CURRENCY_DECIMALS.get(currency_code, 2)
            precision = Decimal(f"0.{'0' * decimals}") if decimals > 0 else Decimal("1")
            return amount.quantize(precision, rounding=ROUND_HALF_UP)

        @staticmethod
        def to_decimal(value) -> Decimal:
            """Safely convert any value to Decimal.

            Always converts via string to avoid float precision issues.
            """
            if isinstance(value, Decimal):
                return value
            return Decimal(str(value))
    ```
